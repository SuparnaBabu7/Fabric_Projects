
let Base = materialize(
    AirQualityData
    | extend
        ReadingTime = todatetime(
            strcat(
                substring(tostring(last_update), 6, 4), "-",
                substring(tostring(last_update), 3, 2), "-",
                substring(tostring(last_update), 0, 2), "T",
                substring(tostring(last_update), 11, 8)
            )
        ),
        CityName = trim(" ", tostring(city)),
        Pollutant = toupper(trim(" ", tostring(pollutant_id))),
        Concentration = todouble(avg_value)
    | where isnotnull(ReadingTime) and isnotnull(Concentration)
);
Base
| where Pollutant == "PM2.5"
| where CityName in~ (
    "Delhi",
    "Mumbai",
    "Hyderabad",
    "Chennai",
    "Kolkata",
    "Vijayawada"
)
| summarize arg_max(ReadingTime, *) by CityName
| extend PM25SubIndex = case(
    Concentration <= 30,  Concentration * 50.0 / 30.0,
    Concentration <= 60,  51.0 + (Concentration - 31.0) * 49.0 / 29.0,
    Concentration <= 90,  101.0 + (Concentration - 61.0) * 99.0 / 29.0,
    Concentration <= 120, 201.0 + (Concentration - 91.0) * 99.0 / 29.0,
    Concentration <= 250, 301.0 + (Concentration - 121.0) * 99.0 / 129.0,
    401.0 + min_of(Concentration, 500.0) - 251.0
)
| project
    CityName,
    Station = station,
    ReadingTime,
    PM25 = Concentration,
    PM25SubIndex = toint(round(min_of(PM25SubIndex, 500.0), 0)),
    Latitude = todouble(latitude),
    Longitude = todouble(longitude)
| order by PM25SubIndex desc
